**Shader Algorithm:**
```wgsl
fn iterate_lemon(c: vec2<f32>, max_iter: u32, convergence_exp: f32, denom_power: f32) -> u32 {
    let threshold = pow(10.0, -convergence_exp);
    var z = c;
    let z0 = c;
    
    for (var i: u32 = 0u; i < max_iter; i++) {
        let z_sq = complex_mul(z, z);
        let one = vec2<f32>(1.0, 0.0);
        
        // numerator = z0 * z^2 * (z^2 + 1)
        let z_sq_plus_1 = z_sq + one;
        let numerator = complex_mul(complex_mul(z0, z_sq), z_sq_plus_1);
        
        // denominator = (z^2 - 1)^k
        let denom_base = z_sq - one;
        let denominator = complex_pow(denom_base, denom_power);
        
        // Check for near-zero denominator
        if (dot(denominator, denominator) < 1e-30) {
            return i;
        }
        
        let z_next = complex_div(numerator, denominator);
        
        // Check for NaN/Inf
        if (!is_finite(z_next)) {
            return i;
        }
        
        // Check convergence
        let delta = z_next - z;
        if (length(delta) < threshold) {
            return i;
        }
        
        // Check for escape (divergence safety)
        if (dot(z_next, z_next) > 1e30) {
            return i;
        }
        
        z = z_next;
    }
    
    return max_iter;
}
```

**Key Differences from Escape-Time Fractals:**
- Uses **convergence** (delta below threshold) instead of escape (magnitude above threshold)
- Requires complex division helper function
- Requires complex power function for denominator (k parameter)
- Has singularity handling in the iteration loop
- The convergence_exp parameter controls precision vs. iteration count tradeoff
- The denom_power parameter (k) gives different fractal shapes: k=2 is canonical, k=1 is "typo variant"
